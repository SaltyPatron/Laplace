name: Live chat forward-pass profile

on:
  push:
    branches: [fix/live-chat-forward-pass]

permissions:
  contents: read

concurrency:
  group: laplace-shared-workspace
  cancel-in-progress: false

env:
  LAPLACE_PG_PREFIX: ${{ vars.LAPLACE_PG_PREFIX || '/opt/laplace/pgsql-18' }}
  PGDATABASE: laplace

jobs:
  profile:
    runs-on: [self-hosted, laplace]
    timeout-minutes: 15
    steps:
      - name: Checkout exact diagnostic revision
        env:
          TARGET_SHA: ${{ github.sha }}
        run: |
          git fetch --no-tags --prune --depth=1 origin "$TARGET_SHA"
          git checkout --force "$TARGET_SHA"
      - uses: ./.github/actions/setup-laplace-env
      - name: Measure deployed dog forward-pass stages
        shell: bash
        run: |
          set -euo pipefail
          export PGHOST=/var/run/postgresql PGUSER=laplace_admin PGDATABASE=laplace
          export PGOPTIONS='-c statement_timeout=15000ms'
          "$LAPLACE_PG_PREFIX/bin/psql" -X -w -v ON_ERROR_STOP=1 <<'SQL'
          \timing on
          \echo CHAT_PROFILE prompt_state
          SELECT count(*) FROM converse.prompt_state('dog');
          \echo CHAT_PROFILE prompt_coherence
          SELECT count(*) FROM converse.prompt_coherence('dog');
          \echo CHAT_PROFILE orient_topic
          SELECT count(*) FROM converse.orient_topic('dog', NULL);
          \echo CHAT_PROFILE route
          SELECT count(*) FROM generation.forward_route_trace('dog', 2, 8, NULL);
          \echo CHAT_PROFILE frontier
          SELECT count(*) FROM generation.forward_frontier('dog', 2, 8, NULL);
          \echo CHAT_PROFILE proposal_word
          SELECT count(*) FROM generation.trajectory_continuations(
              ARRAY[laplace.word_id('dog')]::bytea[], NULL);

          \echo CHAT_PROFILE materialize_step1_operands
          CREATE TEMP TABLE _chat_profile_input AS
          SELECT
              ARRAY(SELECT t.object_id
                      FROM generation.trajectory_continuations(
                               ARRAY[laplace.word_id('dog')]::bytea[], NULL) t) AS candidates,
              ARRAY(SELECT f.entity
                      FROM generation.forward_frontier('dog', 2, 8, NULL) f) AS frontier;
          \echo CHAT_PROFILE steer_input_sizes
          SELECT cardinality(candidates) AS candidates, cardinality(frontier) AS frontier
          FROM _chat_profile_input;
          \echo CHAT_PROFILE steer_word
          SELECT count(*)
          FROM _chat_profile_input i
          CROSS JOIN LATERAL generation.steer_candidates(i.candidates, i.frontier);

          \echo CHAT_PROFILE materialize_first_election
          CREATE TEMP TABLE _chat_profile_first AS
          WITH i AS (SELECT * FROM _chat_profile_input)
          SELECT w.entity
          FROM i CROSS JOIN LATERAL generation.forward_walk_continuations(
              ARRAY[laplace.word_id('dog')]::bytea[], 1, 5, 0.6, 10, 42, i.frontier) w
          WHERE w.step = 1;
          \echo CHAT_PROFILE first_entity
          SELECT encode(entity, 'hex') FROM _chat_profile_first;
          \echo CHAT_PROFILE proposal_bigram
          SELECT count(*) FROM generation.trajectory_continuations(
              ARRAY[laplace.word_id('dog'), (SELECT entity FROM _chat_profile_first)]::bytea[], NULL);
          \echo CHAT_PROFILE proposal_first_unigram
          SELECT count(*) FROM generation.trajectory_continuations(
              ARRAY[(SELECT entity FROM _chat_profile_first)]::bytea[], NULL);

          \echo CHAT_PROFILE forward_1
          SELECT count(*) FROM generation.forward_text('dog', 1, 5, 0.6, 10, 42, 2, 8, NULL);
          \echo CHAT_PROFILE forward_2
          SELECT count(*) FROM generation.forward_text('dog', 2, 5, 0.6, 10, 42, 2, 8, NULL);
          \echo CHAT_PROFILE forward_4
          SELECT count(*) FROM generation.forward_text('dog', 4, 5, 0.6, 10, 42, 2, 8, NULL);
          \echo CHAT_PROFILE forward_8
          SELECT count(*) FROM generation.forward_text('dog', 8, 5, 0.6, 10, 42, 2, 8, NULL);
          SQL
